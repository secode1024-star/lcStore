## 问题10：异步任务监控和重试机制

### 10.1 异步任务状态管理

#### 方案：使用Redis记录异步任务状态

```java
// AsyncTaskService.java
@Service
public class AsyncTaskService {
    
    @Autowired
    private RedisUtil redisUtil;
    
    private static final String TASK_KEY_PREFIX = "async_task:";
    private static final int TASK_EXPIRE_TIME = 3600; // 1小时
    
    /**
     * 记录任务开始
     */
    public void recordTaskStart(String taskType, Integer relatedId) {
        String key = TASK_KEY_PREFIX + taskType + ":" + relatedId;
        
        Map<String, Object> taskInfo = new HashMap<>();
        taskInfo.put("status", "running");
        taskInfo.put("startTime", System.currentTimeMillis());
        taskInfo.put("taskType", taskType);
        taskInfo.put("relatedId", relatedId);
        
        redisUtil.hmset(key, taskInfo, TASK_EXPIRE_TIME);
    }
    
    /**
     * 记录任务完成
     */
    public void recordTaskSuccess(String taskType, Integer relatedId) {
        String key = TASK_KEY_PREFIX + taskType + ":" + relatedId;
        
        Map<String, Object> taskInfo = new HashMap<>();
        taskInfo.put("status", "success");
        taskInfo.put("endTime", System.currentTimeMillis());
        
        redisUtil.hmset(key, taskInfo, TASK_EXPIRE_TIME);
    }
    
    /**
     * 记录任务失败
     */
    public void recordTaskFailure(String taskType, Integer relatedId, String errorMessage) {
        String key = TASK_KEY_PREFIX + taskType + ":" + relatedId;
        
        Map<String, Object> taskInfo = new HashMap<>();
        taskInfo.put("status", "failed");
        taskInfo.put("endTime", System.currentTimeMillis());
        taskInfo.put("errorMessage", errorMessage);
        
        redisUtil.hmset(key, taskInfo, TASK_EXPIRE_TIME);
    }
    
    /**
     * 查询任务状态
     */
    public Map<String, Object> getTaskStatus(String taskType, Integer relatedId) {
        String key = TASK_KEY_PREFIX + taskType + ":" + relatedId;
        return redisUtil.hmget(key);
    }
}
```

### 10.2 翻译任务监控

```java
// TranslationService.java
@Async
public void translateProduct(Integer productId) {
    // 1. 记录任务开始
    asyncTaskService.recordTaskStart("product_translation", productId);
    
    try {
        // 2. 执行翻译
        translateProductBasicInfo(productId);
        translateProductDescription(productId);
        imageTranslationService.translateProductImages(productId);
        
        // 3. 更新商品翻译状态
        updateProductTranslationStatus(productId, 2, null);
        
        // 4. 记录任务成功
        asyncTaskService.recordTaskSuccess("product_translation", productId);
        
        // 5. 发送通知（站内消息）
        notifyMerchant(productId, "商品翻译完成", "您的商品已翻译完成，等待平台审核");
        
    } catch (Exception e) {
        // 6. 记录任务失败
        asyncTaskService.recordTaskFailure("product_translation", productId, e.getMessage());
        
        // 7. 更新商品翻译状态
        updateProductTranslationStatus(productId, 3, e.getMessage());
        
        // 8. 发送失败通知
        notifyMerchant(productId, "商品翻译失败", e.getMessage());
        
        log.error("商品翻译失败: productId={}, error={}", productId, e.getMessage(), e);
    }
}
```

### 10.3 定时检查超时任务

```java
// TranslationMonitorTask.java
@Component
public class TranslationMonitorTask {
    
    @Autowired
    private AsyncTaskService asyncTaskService;
    
    @Autowired
    private StoreProductService storeProductService;
    
    /**
     * 每5分钟检查一次超时任务
     */
    @Scheduled(cron = "0 */5 * * * ?")
    public void checkTimeoutTasks() {
        // 1. 查询翻译中状态超过10分钟的商品
        List<StoreProduct> products = storeProductService.list(
            new LambdaQueryWrapper<StoreProduct>()
                .eq(StoreProduct::getTranslationStatus, 1) // 翻译中
                .lt(StoreProduct::getLastTranslationTime, 
                    DateUtil.addMinutes(new Date(), -10))
        );
        
        for (StoreProduct product : products) {
            // 2. 检查Redis中的任务状态
            Map<String, Object> taskStatus = 
                asyncTaskService.getTaskStatus("product_translation", product.getId());
            
            if (taskStatus == null || "running".equals(taskStatus.get("status"))) {
                // 3. 任务超时，标记为失败
                log.warn("商品翻译任务超时: productId={}", product.getId());
                
                storeProductService.updateTranslationStatus(
                    product.getId(), 3, "翻译任务超时");
                
                asyncTaskService.recordTaskFailure(
                    "product_translation", product.getId(), "任务超时");
            }
        }
    }
}
```

### 10.4 任务重试机制

```java
// TranslationService.java

/**
 * 重试翻译（带重试次数限制）
 */
public void retryTranslation(Integer productId) {
    StoreProduct product = storeProductService.getById(productId);
    
    // 1. 检查翻译状态
    if (product.getTranslationStatus() != 3) {
        throw new BusinessException("只有翻译失败的商品才能重试");
    }
    
    // 2. 检查重试次数（从Redis获取）
    String retryKey = "translation_retry:" + productId;
    Integer retryCount = redisUtil.get(retryKey);
    if (retryCount == null) {
        retryCount = 0;
    }
    
    // 3. 限制重试次数（最多3次）
    if (retryCount >= 3) {
        throw new BusinessException("重试次数已达上限，请联系管理员");
    }
    
    // 4. 增加重试次数
    redisUtil.set(retryKey, retryCount + 1, 86400); // 24小时过期
    
    // 5. 重置翻译状态
    product.setTranslationStatus(1); // 翻译中
    product.setTranslationError("");
    product.setLastTranslationTime(new Date());
    storeProductService.updateById(product);
    
    // 6. 重新触发翻译
    translateProduct(productId);
}
```

### 11.3 VO转换工具类

```java
// MultiLanguageVOConverter.java
@Component
public class MultiLanguageVOConverter {
    
    @Autowired
    private ProductCategoryService categoryService;
    
    /**
     * 转换商品实体为VO（根据当前语言）
     */
    public StoreProductVO convertProductToVO(StoreProduct product) {
        String language = LanguageContext.get();
        
        StoreProductVO vo = new StoreProductVO();
        vo.setId(product.getId());
        vo.setMerId(product.getMerId());
        vo.setCategoryId(product.getCategoryId());
        
        // 根据语言返回对应字段
        switch (language) {
            case "en":
                vo.setStoreName(product.getStoreNameEn());
                vo.setStoreInfo(product.getStoreInfoEn());
                vo.setKeyword(product.getKeywordEn());
                vo.setSliderImages(parseImageUrls(product.getSliderImageEn()));
                vo.setDetailImages(parseImageUrls(product.getDetailImagesEn()));
                vo.setDescription(getProductDescription(product.getId(), "en"));
                break;
            case "ru":
                vo.setStoreName(product.getStoreNameRu());
                vo.setStoreInfo(product.getStoreInfoRu());
                vo.setKeyword(product.getKeywordRu());
                vo.setSliderImages(parseImageUrls(product.getSliderImageRu()));
                vo.setDetailImages(parseImageUrls(product.getDetailImagesRu()));
                vo.setDescription(getProductDescription(product.getId(), "ru"));
                break;
            case "ar":
                vo.setStoreName(product.getStoreNameAr());
                vo.setStoreInfo(product.getStoreInfoAr());
                vo.setKeyword(product.getKeywordAr());
                vo.setSliderImages(parseImageUrls(product.getSliderImageAr()));
                vo.setDetailImages(parseImageUrls(product.getDetailImagesAr()));
                vo.setDescription(getProductDescription(product.getId(), "ar"));
                break;
            default: // zh
                vo.setStoreName(product.getStoreName());
                vo.setStoreInfo(product.getStoreInfo());
                vo.setKeyword(product.getKeyword());
                vo.setSliderImages(parseImageUrls(product.getSliderImage()));
                vo.setDetailImages(parseImageUrls(product.getDetailImages()));
                vo.setDescription(getProductDescription(product.getId(), "zh"));
        }
        
        // 价格字段（所有语言都返回）
        vo.setPrice(product.getPrice());
        vo.setPriceUsd(product.getPriceUsd());
        vo.setOtPrice(product.getOtPrice());
        vo.setOtPriceUsd(product.getOtPriceUsd());
        
        // 分类名称（根据语言返回）
        vo.setCategoryName(getCategoryName(product.getCategoryId(), language));
        
        vo.setCreateTime(product.getCreateTime());
        
        return vo;
    }
    
    /**
     * 获取商品详情（根据语言）
     */
    private String getProductDescription(Integer productId, String language) {
        StoreProductDescription description = 
            productDescriptionService.getByProductId(productId);
        
        if (description == null) {
            return "";
        }
        
        switch (language) {
            case "en":
                return description.getDescriptionEn();
            case "ru":
                return description.getDescriptionRu();
            case "ar":
                return description.getDescriptionAr();
            default:
                return description.getDescription();
        }
    }
    
    /**
     * 获取分类名称（根据语言）
     */
    private String getCategoryName(Integer categoryId, String language) {
        ProductCategory category = categoryService.getById(categoryId);
        
        if (category == null) {
            return "";
        }
        
        switch (language) {
            case "en":
                return category.getNameEn();
            case "ru":
                return category.getNameRu();
            case "ar":
                return category.getNameAr();
            default:
                return category.getName();
        }
    }
    
    /**
     * 批量转换商品列表
     */
    public List<StoreProductVO> convertProductListToVO(List<StoreProduct> products) {
        return products.stream()
            .map(this::convertProductToVO)
            .collect(Collectors.toList());
    }
}
```

## 问题12：翻译缓存策略

### 12.1 缓存设计原则
1. **文本翻译缓存** - 相同文本不重复翻译
2. **图片翻译不缓存** - 每张图片都是独立的，不适合缓存
3. **缓存失效策略** - 永久缓存，除非手动清除
4. **缓存键设计** - `translation:{sourceLanguage}:{targetLanguage}:{md5(content)}`

### 12.2 Translation表作为缓存

```java
// TranslationService.java

/**
 * 翻译文本（带缓存）
 */
public String translateText(String text, String sourceLanguage, String targetLanguage) {
    // 1. 检查缓存
    String cachedTranslation = getFromCache(text, sourceLanguage, targetLanguage);
    if (cachedTranslation != null) {
        log.info("使用缓存翻译: text={}, source={}, target={}", 
            text.substring(0, Math.min(50, text.length())), 
            sourceLanguage, targetLanguage);
        return cachedTranslation;
    }
    
    // 2. 调用大模型API翻译
    String translated = callAiModelTranslate(text, sourceLanguage, targetLanguage);
    
    // 3. 保存到缓存
    saveToCache(text, sourceLanguage, targetLanguage, translated);
    
    return translated;
}

/**
 * 从缓存获取翻译
 */
private String getFromCache(String text, String sourceLanguage, String targetLanguage) {
    // 计算文本MD5
    String contentMd5 = DigestUtils.md5Hex(text);
    
    // 查询缓存表
    Translation cached = translationMapper.selectOne(
        new LambdaQueryWrapper<Translation>()
            .eq(Translation::getSourceLanguage, sourceLanguage)
            .eq(Translation::getTargetLanguage, targetLanguage)
            .eq(Translation::getOriginalContent, contentMd5)
            .eq(Translation::getStatus, 1)
            .last("LIMIT 1")
    );
    
    return cached != null ? cached.getTranslatedContent() : null;
}

/**
 * 保存到缓存
 */
private void saveToCache(String text, String sourceLanguage, 
                        String targetLanguage, String translated) {
    String contentMd5 = DigestUtils.md5Hex(text);
    
    Translation record = new Translation();
    record.setRelatedType("cache");
    record.setRelatedId(0);
    record.setTranslationType("text");
    record.setOriginalContent(contentMd5);
    record.setSourceLanguage(sourceLanguage);
    record.setTargetLanguage(targetLanguage);
    record.setTranslatedContent(translated);
    record.setTokenCount(text.length());
    record.setCost(calculateCost("text", text.length()));
    record.setModelName(getModelName("text"));
    record.setStatus(1);
    record.setCreateTime(new Date());
    
    translationMapper.insert(record);
}
```

### 12.3 Redis二级缓存

```java
// TranslationService.java

/**
 * 翻译文本（带Redis二级缓存）
 */
public String translateTextWithRedisCache(String text, 
                                         String sourceLanguage, 
                                         String targetLanguage) {
    // 1. 检查Redis缓存
    String cacheKey = String.format("translation:%s:%s:%s", 
        sourceLanguage, targetLanguage, DigestUtils.md5Hex(text));
    
    String cachedInRedis = redisUtil.get(cacheKey);
    if (cachedInRedis != null) {
        return cachedInRedis;
    }
    
    // 2. 检查数据库缓存
    String cachedInDb = getFromCache(text, sourceLanguage, targetLanguage);
    if (cachedInDb != null) {
        // 写入Redis缓存
        redisUtil.set(cacheKey, cachedInDb, 86400); // 24小时
        return cachedInDb;
    }
    
    // 3. 调用大模型API翻译
    String translated = callAiModelTranslate(text, sourceLanguage, targetLanguage);
    
    // 4. 保存到数据库缓存
    saveToCache(text, sourceLanguage, targetLanguage, translated);
    
    // 5. 保存到Redis缓存
    redisUtil.set(cacheKey, translated, 86400);
    
    return translated;
}
```

### 12.4 缓存命中率统计

```java
// TranslationService.java

/**
 * 统计缓存命中率
 */
public TranslationCacheStatisticsVO getCacheStatistics(Date startDate, Date endDate) {
    // 查询所有翻译记录
    List<Translation> allRecords = translationMapper.selectList(
        new LambdaQueryWrapper<Translation>()
            .between(Translation::getCreateTime, startDate, endDate)
            .eq(Translation::getTranslationType, "text")
    );
    
    // 查询缓存记录（related_type = 'cache'）
    List<Translation> cacheRecords = translationMapper.selectList(
        new LambdaQueryWrapper<Translation>()
            .between(Translation::getCreateTime, startDate, endDate)
            .eq(Translation::getTranslationType, "text")
            .eq(Translation::getRelatedType, "cache")
    );
    
    TranslationCacheStatisticsVO vo = new TranslationCacheStatisticsVO();
    vo.setTotalTranslations(allRecords.size());
    vo.setCachedTranslations(cacheRecords.size());
    
    // 计算命中率
    if (allRecords.size() > 0) {
        double hitRate = (double) cacheRecords.size() / allRecords.size() * 100;
        vo.setHitRate(BigDecimal.valueOf(hitRate).setScale(2, RoundingMode.HALF_UP));
    } else {
        vo.setHitRate(BigDecimal.ZERO);
    }
    
    // 计算节省的成本
    BigDecimal savedCost = cacheRecords.stream()
        .map(Translation::getCost)
        .reduce(BigDecimal.ZERO, BigDecimal::add);
    vo.setSavedCost(savedCost);
    
    return vo;
}
```

```java
public void cleanExpiredCache() {
```

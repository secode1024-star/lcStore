# 设计方案补充文档 - 翻译机制部分

## 问题4：翻译触发时机详细设计

### 4.1 商品上架时翻译

#### 触发点
```java
// StoreProductService.java
public void onSale(Integer productId) {
    // 1. 检查商品状态
    StoreProduct product = getById(productId);
    if (product.getIsShow()) {
        throw new BusinessException("商品已上架");
    }
    
    // 2. 触发翻译（异步）
    translationService.translateProduct(productId);
    
    // 3. 更新商品状态为"翻译中"
    product.setTranslationStatus(1); // 翻译中
    product.setIsShow(false); // 暂不上架
    updateById(product);
}
```

#### 翻译完成回调
```java
// TranslationService.java
@Async
public void translateProduct(Integer productId) {
    try {
        // 1. 翻译商品基本信息（名称、简介、关键词）
        translateProductBasicInfo(productId);
        
        // 2. 翻译商品详情
        translateProductDescription(productId);
        
        // 3. 翻译商品图片
        imageTranslationService.translateProductImages(productId);
        
        // 4. 更新翻译状态为"翻译完成"
        updateProductTranslationStatus(productId, 2, null);
        
        // 5. 商品进入待审核状态
        storeProductService.updateAuditStatus(productId, 0); // 待审核
        
    } catch (Exception e) {
        // 翻译失败，更新状态
        updateProductTranslationStatus(productId, 3, e.getMessage());
    }
}
```

### 4.2 分类修改时翻译

#### 触发点
```java
// CategoryService.java
public void update(ProductCategory category) {
    // 1. 更新分类信息
    updateById(category);
    
    // 2. 检查名称是否修改
    ProductCategory oldCategory = getById(category.getId());
    if (!oldCategory.getName().equals(category.getName())) {
        // 3. 触发翻译（异步）
        translationService.translateCategory(category.getId());
    }
}
```

#### 翻译实现
```java
// TranslationService.java
@Async
public void translateCategory(Integer categoryId) {
    try {
        ProductCategory category = categoryService.getById(categoryId);
        
        // 1. 翻译分类名称
        String nameEn = translateText(category.getName(), "zh", "en");
        String nameRu = translateText(category.getName(), "zh", "ru");
        String nameAr = translateText(category.getName(), "zh", "ar");
        
        // 2. 更新分类多语言字段
        category.setNameEn(nameEn);
        category.setNameRu(nameRu);
        category.setNameAr(nameAr);
        category.setTranslationStatus(2); // 翻译完成
        category.setLastTranslationTime(new Date());
        
        categoryService.updateById(category);
        
        // 3. 清除分类缓存
        redisUtil.delete("category:" + categoryId);
        
    } catch (Exception e) {
        // 翻译失败处理
        updateCategoryTranslationStatus(categoryId, 3, e.getMessage());
    }
}
```

#### 监听器实现
```java
// SystemConfigTranslationListener.java
@Component
public class SystemConfigTranslationListener {
    
    @Autowired
    private TranslationService translationService;
    
    // 需要翻译的配置项
    private static final Set<String> TRANSLATABLE_CONFIGS = new HashSet<>(Arrays.asList(
        "site_name",           // 网站名称
        "site_keywords",       // SEO关键词
        "site_description",    // SEO描述
        "footer_copyright",    // 页脚版权信息
        "footer_about"         // 页脚关于我们
    ));
    
    @EventListener
    @Async
    public void onConfigUpdate(SystemConfigUpdateEvent event) {
        String key = event.getKey();
        
        // 检查是否需要翻译
        if (TRANSLATABLE_CONFIGS.contains(key)) {
            translationService.translateSystemConfig(key, event.getValue());
        }
    }
}
```

#### 翻译实现
```java
// TranslationService.java
@Async
public void translateSystemConfig(String key, String value) {
    try {
        // 1. 翻译配置值
        String valueEn = translateText(value, "zh", "en");
        String valueRu = translateText(value, "zh", "ru");
        String valueAr = translateText(value, "zh", "ar");
        
        // 2. 保存多语言配置
        systemConfigService.saveMultiLanguage(key, valueEn, valueRu, valueAr);

        redisUtil.delete("system_config:" + key);
        
    } catch (Exception e) {
        log.error("系统配置翻译失败: key={}, error={}", key, e.getMessage());
    }
}
```

## 问题5：翻译失败处理机制

### 5.1 翻译失败的场景
1. **API调用失败** - 大模型API不可用、超时、限流
2. **部分翻译失败** - 某个语言翻译成功，其他语言失败
3. **图片翻译失败** - 图片下载失败、翻译失败、上传OSS失败
4. **翻译超时** - 翻译任务执行时间过长

### 5.2 处理策略

#### 策略1：商品翻译失败
```java
// 翻译失败后的处理
public void handleProductTranslationFailure(Integer productId, String errorMessage) {
    StoreProduct product = getById(productId);
    
    // 1. 更新翻译状态为"翻译失败"
    product.setTranslationStatus(3);
    product.setTranslationError(errorMessage);
    product.setLastTranslationTime(new Date());
    
    // 2. 商品保持下架状态
    product.setIsShow(false);
    
    // 3. 审核状态重置为"未提交"
    product.setAuditStatus(null);
    
    updateById(product);
    
    // 4. 通知商家（站内消息）
    notifyMerchant(product.getMerId(), 
        "商品翻译失败", 
        String.format("商品【%s】翻译失败，请重新上架。错误信息：%s", 
            product.getStoreName(), errorMessage));
}
```

#### 策略2：部分翻译失败
```java
// 部分语言翻译失败的处理
public void handlePartialTranslationFailure(Integer productId, 
                                            Map<String, String> successLanguages,
                                            Map<String, String> failedLanguages) {
    // 1. 保存成功的翻译结果
    saveSuccessfulTranslations(productId, successLanguages);
    
    // 2. 记录失败的语言
    String errorMessage = "部分语言翻译失败：" + 
        failedLanguages.entrySet().stream()
            .map(e -> e.getKey() + "(" + e.getValue() + ")")
            .collect(Collectors.joining(", "));
    
    // 3. 更新翻译状态为"翻译失败"
    updateProductTranslationStatus(productId, 3, errorMessage);
    
    // 4. 商品不允许上架
    // 原因：必须所有语言都翻译成功才能上架
}
```

#### 策略3：图片翻译失败
```java
// 图片翻译失败的处理
public void handleImageTranslationFailure(Integer productId, 
                                         String imageType, 
                                         Integer imageIndex,
                                         String errorMessage) {
    // 1. 记录失败的图片
    ProductImageTranslation record = new ProductImageTranslation();
    record.setProductId(productId);
    record.setImageType(imageType);
    record.setImageIndex(imageIndex);
    record.setTranslationStatus(3); // 失败
    record.setErrorMessage(errorMessage);
    productImageTranslationMapper.insert(record);
    
    // 2. 整个商品翻译标记为失败
    String fullError = String.format("图片翻译失败：%s第%d张，错误：%s", 
        imageType, imageIndex, errorMessage);
    handleProductTranslationFailure(productId, fullError);
}
```

```java
// 异步任务超时配置
@Configuration
public class AsyncConfig implements AsyncConfigurer {
    
    @Override
    public Executor getAsyncExecutor() {
        ThreadPoolTaskExecutor executor = new ThreadPoolTaskExecutor();
        executor.setCorePoolSize(5);
        executor.setMaxPoolSize(10);
        executor.setQueueCapacity(100);
        executor.setThreadNamePrefix("translation-");
        
        // 设置超时时间：10分钟
        executor.setAwaitTerminationSeconds(600);
        
        executor.initialize();
        return executor;
    }
}

// 翻译任务超时检测
@Scheduled(cron = "0 */5 * * * ?") // 每5分钟检查一次
public void checkTranslationTimeout() {
    // 查询翻译中状态超过10分钟的商品
    List<StoreProduct> timeoutProducts = storeProductMapper.selectList(
        new LambdaQueryWrapper<StoreProduct>()
            .eq(StoreProduct::getTranslationStatus, 1) // 翻译中
            .lt(StoreProduct::getLastTranslationTime, 
                DateUtil.addMinutes(new Date(), -10))
    );
    
    // 标记为翻译失败
    for (StoreProduct product : timeoutProducts) {
        handleProductTranslationFailure(product.getId(), "翻译超时");
    }
}
```

### 5.3 重新翻译机制
```java
// 商家端：重新翻译接口
public void retryTranslation(Integer productId) {
    StoreProduct product = getById(productId);
    
    // 1. 检查商品状态
    if (product.getTranslationStatus() != 3) {
        throw new BusinessException("只有翻译失败的商品才能重新翻译");
    }
    
    // 2. 清空之前的翻译错误信息
    product.setTranslationError("");
    product.setTranslationStatus(1); // 翻译中
    product.setLastTranslationTime(new Date());
    updateById(product);
    
    // 3. 重新触发翻译
    translationService.translateProduct(productId);
}
```

## 问题6：图片翻译详细流程

### 6.1 完整流程图
```
商品上架 
  ↓
触发图片翻译
  ↓
遍历所有图片（轮播图 + 详情图）
  ↓
对每张图片执行：
  1. 下载原始图片到临时目录
  2. 调用大模型API翻译图片（生成3个语言版本）
  3. 上传翻译后的图片到OSS（按语言分目录）
  4. 记录翻译结果到 eb_product_image_translation 表
  5. 删除临时文件
  ↓
所有图片翻译完成
  ↓
更新商品表的多语言图片字段
  ↓
翻译完成
```

### 6.2 代码实现
```java
// ImageTranslationService.java
@Service
public class ImageTranslationServiceImpl implements ImageTranslationService {
    
    @Autowired
    private OssService ossService;
    
    @Autowired
    private AiModelService aiModelService;
    
    @Autowired
    private ProductImageTranslationMapper imageTranslationMapper;
    
    /**
     * 翻译商品所有图片
     */
    @Override
    @Async
    public void translateProductImages(Integer productId) {
        StoreProduct product = storeProductService.getById(productId);
        
        try {
            // 1. 翻译轮播图
            List<String> sliderImages = parseImageUrls(product.getSliderImage());
            Map<String, List<String>> translatedSliderImages = 
                translateImageList(productId, "slider", sliderImages);
            
            // 2. 翻译详情图
            List<String> detailImages = parseImageUrls(product.getDetailImages());
            Map<String, List<String>> translatedDetailImages = 
                translateImageList(productId, "detail", detailImages);
            
            // 3. 更新商品表的多语言图片字段
            updateProductImageFields(productId, 
                translatedSliderImages, translatedDetailImages);
            
        } catch (Exception e) {
            throw new BusinessException("图片翻译失败：" + e.getMessage());
        }
    }
    
    /**
     * 翻译图片列表
     */
    private Map<String, List<String>> translateImageList(
            Integer productId, String imageType, List<String> imageUrls) {
        
        Map<String, List<String>> result = new HashMap<>();
        result.put("en", new ArrayList<>());
        result.put("ru", new ArrayList<>());
        result.put("ar", new ArrayList<>());
        
        for (int i = 0; i < imageUrls.size(); i++) {
            String originalUrl = imageUrls.get(i);
            
            // 翻译单张图片
            Map<String, String> translated = 
                translateSingleImage(productId, imageType, i, originalUrl);
            
            result.get("en").add(translated.get("en"));
            result.get("ru").add(translated.get("ru"));
            result.get("ar").add(translated.get("ar"));
        }
        
        return result;
    }
    
    /**
     * 翻译单张图片
     */
    private Map<String, String> translateSingleImage(
            Integer productId, String imageType, 
            Integer imageIndex, String originalUrl) {
        
        Map<String, String> result = new HashMap<>();
        File tempFile = null;
        
        try {
            // 1. 下载原始图片到临时目录
            tempFile = downloadImage(originalUrl);
            
            // 2. 调用大模型API翻译图片（3个语言）
            Map<String, byte[]> translatedImages = 
                aiModelService.translateImage(tempFile, 
                    Arrays.asList("en", "ru", "ar"));
            
            // 3. 上传翻译后的图片到OSS
            for (Map.Entry<String, byte[]> entry : translatedImages.entrySet()) {
                String language = entry.getKey();
                byte[] imageData = entry.getValue();
                
                // OSS路径：product/{language}/{productId}/{imageType}_{index}.jpg
                String ossPath = String.format("product/%s/%d/%s_%d.jpg", 
                    language, productId, imageType, imageIndex);
                
                String ossUrl = ossService.upload(ossPath, imageData);
                result.put(language, ossUrl);
            }
            
            // 4. 记录翻译结果
            saveTranslationRecord(productId, imageType, imageIndex, 
                originalUrl, result);
            
            return result;
            
        } catch (Exception e) {
            // 记录失败
            saveFailedRecord(productId, imageType, imageIndex, 
                originalUrl, e.getMessage());
            throw new BusinessException("图片翻译失败：" + e.getMessage());
            
        } finally {
            // 5. 删除临时文件
            if (tempFile != null && tempFile.exists()) {
                tempFile.delete();
            }
        }
    }
}
```

## 2. 分页参数统一规范

### 2.1 分页参数标准

**请求参数**:
| 参数名 | 类型 | 必填 | 默认值 | 说明 |
|--------|------|------|--------|------|
| pageNum | Integer | 否 | 1 | 页码（从1开始） |
| pageSize | Integer | 否 | 10 | 每页数量 |

**响应格式**:
```json
{
  "code": 200,
  "message": "success",
  "data": {
    "list": [],           // 数据列表
    "total": 100,         // 总记录数
    "pageNum": 1,         // 当前页码
    "pageSize": 10,       // 每页数量
    "pages": 10,          // 总页数
    "hasNextPage": true,  // 是否有下一页
    "hasPreviousPage": false  // 是否有上一页
  }
}
```

### 2.3 分页响应类

```java
package com.zbkj.common.response;

import com.github.pagehelper.PageInfo;
import io.swagger.annotations.ApiModel;
import io.swagger.annotations.ApiModelProperty;
import lombok.Data;

import java.util.List;

/**
 * 分页响应类
 */
@Data
@ApiModel(value = "PageResponse", description = "分页响应")
public class PageResponse<T> {

    @ApiModelProperty(value = "数据列表")
    private List<T> list;

    @ApiModelProperty(value = "总记录数")
    private Long total;

    @ApiModelProperty(value = "当前页码")
    private Integer pageNum;

    @ApiModelProperty(value = "每页数量")
    private Integer pageSize;

    @ApiModelProperty(value = "总页数")
    private Integer pages;

    @ApiModelProperty(value = "是否有下一页")
    private Boolean hasNextPage;

    @ApiModelProperty(value = "是否有上一页")
    private Boolean hasPreviousPage;

    /**
     * 从PageInfo构建分页响应
     */
    public static <T> PageResponse<T> of(PageInfo<T> pageInfo) {
        PageResponse<T> response = new PageResponse<>();
        response.setList(pageInfo.getList());
        response.setTotal(pageInfo.getTotal());
        response.setPageNum(pageInfo.getPageNum());
        response.setPageSize(pageInfo.getPageSize());
        response.setPages(pageInfo.getPages());
        response.setHasNextPage(pageInfo.isHasNextPage());
        response.setHasPreviousPage(pageInfo.isHasPreviousPage());
        return response;
    }

    /**
     * 从PageInfo构建分页响应（转换数据类型）
     */
    public static <T, R> PageResponse<R> of(PageInfo<T> pageInfo, List<R> list) {
        PageResponse<R> response = new PageResponse<>();
        response.setList(list);
        response.setTotal(pageInfo.getTotal());
        response.setPageNum(pageInfo.getPageNum());
        response.setPageSize(pageInfo.getPageSize());
        response.setPages(pageInfo.getPages());
        response.setHasNextPage(pageInfo.isHasNextPage());
        response.setHasPreviousPage(pageInfo.isHasPreviousPage());
        return response;
    }
}
```

```java
/**
 * 获取商品列表
 */
@GetMapping("/list")
public CommonResult<PageResponse<ProductResponse>> getProductList(
        @Valid PageRequest pageRequest,
        @RequestParam(required = false) String keyword) {
    
    // 开启分页
    PageHelper.startPage(pageRequest.getPageNum(), pageRequest.getPageSize());
    
    // 查询数据
    List<StoreProduct> list = productService.getList(keyword);
    
    // 构建分页信息
    PageInfo<StoreProduct> pageInfo = new PageInfo<>(list);
    
    // 转换为响应对象
    List<ProductResponse> responseList = list.stream()
        .map(this::convertToResponse)
        .collect(Collectors.toList());
    
    // 返回分页响应
    PageResponse<ProductResponse> pageResponse = PageResponse.of(pageInfo, responseList);
    return CommonResult.success(pageResponse);
}
```

## 3. API接口权限说明

### 3.1 权限级别定义

| 权限级别 | 说明 | 示例 |
|---------|------|------|
| PUBLIC | 公开接口，无需登录 | 商品列表、商品详情、国家列表 |
| USER | 用户接口，需要登录 | 我的询价单、个人信息、地址管理 |
| MERCHANT | 商家接口，需要商家角色 | 商品管理、翻译进度查看 |
| PLATFORM | 平台接口，需要平台管理员角色 | 商品审核、询价管理、商家管理 |

### 3.2 接口权限注解

```java
package com.zbkj.common.annotation;

import java.lang.annotation.*;

/**
 * 接口权限注解
 */
@Target({ElementType.METHOD, ElementType.TYPE})
@Retention(RetentionPolicy.RUNTIME)
@Documented
public @interface RequirePermission {
    
    /**
     * 权限级别
     */
    PermissionLevel value() default PermissionLevel.PUBLIC;
    
    /**
     * 权限级别枚举
     */
    enum PermissionLevel {
        PUBLIC,    // 公开
        USER,      // 用户
        MERCHANT,  // 商家
        PLATFORM   // 平台
    }
}
```

### 3.3 权限拦截器

```java
package com.zbkj.common.interceptor;

import com.zbkj.common.annotation.RequirePermission;
import com.zbkj.common.exception.UnauthorizedException;
import com.zbkj.common.utils.JwtUtil;
import org.springframework.stereotype.Component;
import org.springframework.web.method.HandlerMethod;
import org.springframework.web.servlet.HandlerInterceptor;

import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

/**
 * 权限拦截器
 */
@Component
public class PermissionInterceptor implements HandlerInterceptor {

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        if (!(handler instanceof HandlerMethod)) {
            return true;
        }

        HandlerMethod handlerMethod = (HandlerMethod) handler;
        
        // 获取方法上的权限注解
        RequirePermission annotation = handlerMethod.getMethodAnnotation(RequirePermission.class);
        if (annotation == null) {
            // 获取类上的权限注解
            annotation = handlerMethod.getBeanType().getAnnotation(RequirePermission.class);
        }
        
        // 如果没有权限注解，默认为公开接口
        if (annotation == null) {
            return true;
        }

        RequirePermission.PermissionLevel level = annotation.value();
        
        // 公开接口，直接放行
        if (level == RequirePermission.PermissionLevel.PUBLIC) {
            return true;
        }

        // 获取token
        String token = request.getHeader("Authorization");
        if (token == null || !token.startsWith("Bearer ")) {
            throw new UnauthorizedException("未登录");
        }

        token = token.substring(7);
        
        // 验证token
        if (!JwtUtil.verify(token)) {
            throw new UnauthorizedException("Token无效");
        }

        // 获取用户角色
        String role = JwtUtil.getRole(token);
        
        // 验证权限
        switch (level) {
            case USER:
                // 用户接口，只要登录即可
                break;
            case MERCHANT:
                // 商家接口，需要商家角色
                if (!"merchant".equals(role)) {
                    throw new UnauthorizedException("无权限访问");
                }
                break;
            case PLATFORM:
                // 平台接口，需要平台管理员角色
                if (!"platform".equals(role)) {
                    throw new UnauthorizedException("无权限访问");
                }
                break;
        }

        return true;
    }
}
```

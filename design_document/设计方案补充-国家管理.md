## 3. API接口设计

### 3.1 获取国家列表

**接口路径**: `GET /api/common/country/list`

**接口说明**: 获取国家列表，根据Accept-Language返回对应语言的国家名称

**请求参数**:
| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| isHot | Integer | 否 | 是否只返回热门国家（1-是，0或不传-全部） |

**请求头**:
```
Accept-Language: zh/en/ru/ar
```

**响应示例**:
```json
{
  "code": 200,
  "message": "success",
  "data": [
    {
      "code": "CN",
      "name": "中国",
      "phoneCode": "+86",
      "isHot": true
    },
    {
      "code": "US",
      "name": "美国",
      "phoneCode": "+1",
      "isHot": true
    }
  ]
}
```

### 3.2 根据代码获取国家信息

**接口路径**: `GET /api/common/country/detail/{code}`

**接口说明**: 根据国家代码获取国家详细信息

**路径参数**:
| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| code | String | 是 | 国家代码 |

**响应示例**:
```json
{
  "code": 200,
  "message": "success",
  "data": {
    "code": "CN",
    "name": "中国",
    "phoneCode": "+86",
    "isHot": true
  }
}
```

### 4.1 CountryService接口

package com.zbkj.service.service;

import com.zbkj.common.model.country.Country;
import com.zbkj.common.response.CountryResponse;
import com.baomidou.mybatisplus.extension.service.IService;

import java.util.List;

/**
 * 国家服务接口
 */
public interface CountryService extends IService<Country> {

    /**
     * 获取国家列表（根据语言返回）
     * @param isHot 是否只返回热门国家
     * @param language 语言
     * @return 国家列表

    List<CountryResponse> getCountryList(Integer isHot, String language);

     * 根据代码获取国家信息
     * @param code 国家代码

    /**
     * 根据语言获取国家名称
     * @param country 国家对象
     * @param language 语言
     * @return 国家名称
     */
    String getCountryName(Country country, String language);
}

### 4.2 CountryServiceImpl实现类

```java
package com.zbkj.service.service.impl;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.zbkj.common.model.country.Country;
import com.zbkj.common.response.CountryResponse;
import com.zbkj.service.dao.CountryDao;
import com.zbkj.service.service.CountryService;
import org.springframework.beans.BeanUtils;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.stream.Collectors;

/**
 * 国家服务实现类
 */
@Service
public class CountryServiceImpl extends ServiceImpl<CountryDao, Country> implements CountryService {

    @Override
    public List<CountryResponse> getCountryList(Integer isHot, String language) {
        LambdaQueryWrapper<Country> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(Country::getIsShow, 1);
        if (isHot != null && isHot == 1) {
            wrapper.eq(Country::getIsHot, 1);
        }
        wrapper.orderByAsc(Country::getSort);
        
        List<Country> countryList = list(wrapper);
        
        return countryList.stream().map(country -> {
            CountryResponse response = new CountryResponse();
            response.setCode(country.getCode());
            response.setName(getCountryName(country, language));
            response.setPhoneCode(country.getPhoneCode());
            response.setIsHot(country.getIsHot() == 1);
            return response;
        }).collect(Collectors.toList());
    }

    @Override
    public CountryResponse getCountryByCode(String code, String language) {
        LambdaQueryWrapper<Country> wrapper = new LambdaQueryWrapper<>();
        wrapper.eq(Country::getCode, code);
        wrapper.eq(Country::getIsShow, 1);
        
        Country country = getOne(wrapper);
        if (country == null) {
            return null;
        }
        
        CountryResponse response = new CountryResponse();
        response.setCode(country.getCode());
        response.setName(getCountryName(country, language));
        response.setPhoneCode(country.getPhoneCode());
        response.setIsHot(country.getIsHot() == 1);
        
        return response;
    }

    @Override
    public String getCountryName(Country country, String language) {
        if (country == null) {
            return "";
        }
        
        switch (language) {
            case "en":
                return country.getNameEn();
            case "ru":
                return country.getNameRu();
            case "ar":
                return country.getNameAr();
            case "zh":
            default:
                return country.getNameZh();
        }
    }
}
```

## 5. Controller层设计

### 5.1 CommonController

```java
package com.zbkj.front.controller;

import com.zbkj.common.response.CommonResult;
import com.zbkj.common.response.CountryResponse;
import com.zbkj.service.service.CountryService;
import io.swagger.annotations.Api;
import io.swagger.annotations.ApiOperation;
import io.swagger.annotations.ApiParam;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import javax.servlet.http.HttpServletRequest;
import java.util.List;

/**
 * 公共接口控制器
 */
@Slf4j
@RestController
@RequestMapping("/api/common")
@Api(tags = "公共接口")
public class CommonController {

    @Autowired
    private CountryService countryService;

    @ApiOperation(value = "获取国家列表")
    @GetMapping("/country/list")
    public CommonResult<List<CountryResponse>> getCountryList(
            @ApiParam(value = "是否只返回热门国家") @RequestParam(required = false) Integer isHot,
            HttpServletRequest request) {
        String language = request.getHeader("Accept-Language");
        if (language == null || language.isEmpty()) {
            language = "zh";
        }
        List<CountryResponse> list = countryService.getCountryList(isHot, language);
        return CommonResult.success(list);
    }

    @ApiOperation(value = "根据代码获取国家信息")
    @GetMapping("/country/detail/{code}")
    public CommonResult<CountryResponse> getCountryByCode(
            @ApiParam(value = "国家代码", required = true) @PathVariable String code,
            HttpServletRequest request) {
        String language = request.getHeader("Accept-Language");
        if (language == null || language.isEmpty()) {
            language = "zh";
        }
        CountryResponse country = countryService.getCountryByCode(code, language);
        return CommonResult.success(country);
    }
}
```

## 7. 注意事项

1. **国家代码标准**: 使用ISO 3166-1 alpha-2标准（两位字母代码）
2. **多语言支持**: 所有国家名称都需要提供中英俄阿四种语言
3. **热门国家**: 可以在平台端管理热门国家，方便用户快速选择
4. **排序**: 热门国家排在前面，其他国家按字母顺序排列
5. **缓存**: 国家数据变化不频繁，可以考虑缓存优化
6. **扩展性**: 如需添加更多国家，只需在数据库中插入数据即可

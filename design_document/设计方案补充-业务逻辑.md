```java
// InquiryService.java
public String createInquiry(InquiryRequest request) {
    // 1. 获取当前用户语言
    String userLanguage = LanguageContext.get(); // 从ThreadLocal获取
    
    // 2. 创建询价单主表
    StoreOrder inquiry = new StoreOrder();
    inquiry.setInquiryNo(generateInquiryNo());
    inquiry.setUid(request.getUid());
    inquiry.setUserLanguage(userLanguage);
    // ... 其他字段
    
    // 3. 翻译用户填写的信息为中文（如果用户不是中文）
    if (!"zh".equals(userLanguage)) {
        inquiry.setRealName(request.getRealName()); // 中文翻译
        inquiry.setRealNameOriginal(request.getRealNameOriginal()); // 原文
        inquiry.setUserAddress(translateToChinese(request.getUserAddress()));
        inquiry.setUserAddressOriginal(request.getUserAddress());
        inquiry.setUserRemark(translateToChinese(request.getUserRemark()));
        inquiry.setUserRemarkOriginal(request.getUserRemark());
    } else {
        // 用户本身就是中文，不需要翻译
        inquiry.setRealName(request.getRealName());
        inquiry.setRealNameOriginal(request.getRealName());
        inquiry.setUserAddress(request.getUserAddress());
        inquiry.setUserAddressOriginal(request.getUserAddress());
        inquiry.setUserRemark(request.getUserRemark());
        inquiry.setUserRemarkOriginal(request.getUserRemark());
    }
    
    storeOrderMapper.insert(inquiry);
    
    // 4. 创建询价单详情（商品快照）
    for (InquiryCartItem item : request.getCartItems()) {
        StoreOrderInfo orderInfo = createOrderInfoSnapshot(
            inquiry.getInquiryNo(), 
            item.getProductId(), 
            item.getQuantity(),
            userLanguage
        );
        storeOrderInfoMapper.insert(orderInfo);
    }
    
    return inquiry.getInquiryNo();
}

/**
 * 创建商品快照
 */
private StoreOrderInfo createOrderInfoSnapshot(String inquiryNo, 
                                               Integer productId, 
                                               Integer quantity,
                                               String userLanguage) {
    // 1. 查询商品信息
    StoreProduct product = storeProductService.getById(productId);
    
    // 2. 根据用户语言获取对应的商品信息
    String productNameUser = getProductNameByLanguage(product, userLanguage);
    String imageUser = getProductImageByLanguage(product, userLanguage);
    
    // 3. 创建订单详情
    StoreOrderInfo orderInfo = new StoreOrderInfo();
    orderInfo.setMerOrderNo(inquiryNo);
    orderInfo.setMerId(product.getMerId());
    orderInfo.setProductId(productId);
    
    // 中文信息（平台端查看）
    orderInfo.setProductName(product.getStoreName());
    orderInfo.setImage(product.getImage());
    
    // 用户语言信息（用户查看）
    orderInfo.setProductNameOriginal(productNameUser);
    orderInfo.setImageOriginal(imageUser);
    orderInfo.setLanguage(userLanguage);
    
    // 价格（保存用户看到的价格）
    orderInfo.setPrice(product.getPrice()); // 人民币
    orderInfo.setPayNum(quantity);
    
    return orderInfo;
}

/**
 * 根据语言获取商品名称
 */
private String getProductNameByLanguage(StoreProduct product, String language) {
    switch (language) {
        case "en":
            return product.getStoreNameEn();
        case "ru":
            return product.getStoreNameRu();
        case "ar":
            return product.getStoreNameAr();
        default:
            return product.getStoreName();
    }
}

/**
 * 根据语言获取商品图片（第一张轮播图）
 */
private String getProductImageByLanguage(StoreProduct product, String language) {
    String sliderImages;
    switch (language) {
        case "en":
            sliderImages = product.getSliderImageEn();
            break;
        case "ru":
            sliderImages = product.getSliderImageRu();
            break;
        case "ar":
            sliderImages = product.getSliderImageAr();
            break;
        default:
            sliderImages = product.getSliderImage();
    }
    
    // 取第一张图片
    List<String> images = parseImageUrls(sliderImages);
    return images.isEmpty() ? "" : images.get(0);
}
```

```java
// SystemConfigService.java
public void updateConfig(String key, String value) {
    SystemConfig config = getByKey(key);
    
    // 1. 更新中文配置值
    config.setValue(value);
    updateById(config);
    
    // 2. 检查是否需要翻译
    if (TranslatableConfigKey.isTranslatable(key)) {
        // 3. 触发异步翻译
        translationService.translateSystemConfig(config.getId(), key, value);
    }
    
    // 4. 清除缓存
    redisUtil.delete("system_config:" + key);
}

// TranslationService.java
@Async
public void translateSystemConfig(Integer configId, String key, String value) {
    try {
        // 1. 翻译配置值
        String valueEn = translateText(value, "zh", "en");
        String valueRu = translateText(value, "zh", "ru");
        String valueAr = translateText(value, "zh", "ar");
        
        // 2. 更新配置表
        SystemConfig config = systemConfigService.getById(configId);
        config.setValueEn(valueEn);
        config.setValueRu(valueRu);
        config.setValueAr(valueAr);
        config.setTranslationStatus(2); // 翻译完成
        config.setLastTranslationTime(new Date());
        
        systemConfigService.updateById(config);
        
        // 3. 记录翻译成本
        recordTranslationCost("system_config", configId, "text", 
            value.length(), "zh", Arrays.asList("en", "ru", "ar"));
        
    } catch (Exception e) {
        // 翻译失败
        SystemConfig config = systemConfigService.getById(configId);
        config.setTranslationStatus(3);
        systemConfigService.updateById(config);
        
        log.error("系统配置翻译失败: key={}, error={}", key, e.getMessage());
    }
}
```

### 8.5 C端获取配置（根据语言返回）

```java
// SystemConfigService.java
public String getConfigValue(String key) {
    // 1. 从缓存获取
    String cacheKey = "system_config:" + key;
    String cachedValue = redisUtil.get(cacheKey);
    if (cachedValue != null) {
        return cachedValue;
    }
    
    // 2. 从数据库获取
    SystemConfig config = getByKey(key);
    if (config == null) {
        return "";
    }
    
    // 3. 根据当前语言返回对应值
    String language = LanguageContext.get();
    String value;
    
    switch (language) {
        case "en":
            value = config.getValueEn();
            break;
        case "ru":
            value = config.getValueRu();
            break;
        case "ar":
            value = config.getValueAr();
            break;
        default:
            value = config.getValue();
    }
    
    // 4. 如果多语言值为空，降级到中文
    if (StringUtils.isBlank(value)) {
        value = config.getValue();
    }
    
    // 5. 缓存结果（按语言分别缓存）
    redisUtil.set(cacheKey + ":" + language, value, 3600);
    
    return value;
}
```

```java
// TranslationService.java
public void recordTranslationCost(String relatedType, 
                                  Integer relatedId,
                                  String translationType,
                                  Integer tokenCount,
                                  String sourceLanguage,
                                  List<String> targetLanguages) {
    for (String targetLanguage : targetLanguages) {
        Translation record = new Translation();
        record.setRelatedType(relatedType);
        record.setRelatedId(relatedId);
        record.setTranslationType(translationType);
        record.setTokenCount(tokenCount);
        record.setSourceLanguage(sourceLanguage);
        record.setTargetLanguage(targetLanguage);
        record.setModelName(getModelName(translationType));
        record.setCost(calculateCost(translationType, tokenCount));
        record.setStatus(1);
        record.setCreateTime(new Date());
        
        translationMapper.insert(record);
    }
}

/**
 * 计算翻译成本
 */
private BigDecimal calculateCost(String translationType, Integer tokenCount) {
    // 文本翻译：0.001元/1000 tokens
    // 图片翻译：0.01元/张
    if ("text".equals(translationType)) {
        return new BigDecimal(tokenCount)
            .divide(new BigDecimal(1000), 6, RoundingMode.HALF_UP)
            .multiply(new BigDecimal("0.001"));
    } else {
        return new BigDecimal("0.01");
    }
}
```

```java
// TranslationService.java

/**
 * 查询翻译成本统计（按日期）
 */
public TranslationCostStatisticsVO getCostStatistics(Date startDate, Date endDate) {
    List<Translation> records = translationMapper.selectList(
        new LambdaQueryWrapper<Translation>()
            .between(Translation::getCreateTime, startDate, endDate)
            .eq(Translation::getStatus, 1)
    );
    
    TranslationCostStatisticsVO vo = new TranslationCostStatisticsVO();
    
    // 总成本
    BigDecimal totalCost = records.stream()
        .map(Translation::getCost)
        .reduce(BigDecimal.ZERO, BigDecimal::add);
    vo.setTotalCost(totalCost);
    
    // 按类型统计
    Map<String, BigDecimal> costByType = records.stream()
        .collect(Collectors.groupingBy(
            Translation::getRelatedType,
            Collectors.reducing(BigDecimal.ZERO, 
                Translation::getCost, BigDecimal::add)
        ));
    vo.setCostByType(costByType);
    
    // 按翻译类型统计
    Map<String, BigDecimal> costByTranslationType = records.stream()
        .collect(Collectors.groupingBy(
            Translation::getTranslationType,
            Collectors.reducing(BigDecimal.ZERO, 
                Translation::getCost, BigDecimal::add)
        ));
    vo.setCostByTranslationType(costByTranslationType);
    
    // 总Token数
    Integer totalTokens = records.stream()
        .filter(r -> "text".equals(r.getTranslationType()))
        .mapToInt(Translation::getTokenCount)
        .sum();
    vo.setTotalTokens(totalTokens);
    
    // 总翻译次数
    vo.setTotalCount(records.size());
    
    return vo;
}

/**
 * 查询商品翻译成本

public BigDecimal getProductTranslationCost(Integer productId) {
    List<Translation> records = translationMapper.selectList(
        new LambdaQueryWrapper<Translation>()
            .eq(Translation::getRelatedType, "product")
            .eq(Translation::getRelatedId, productId)
            .eq(Translation::getStatus, 1)
    );
    
    return records.stream()
        .map(Translation::getCost)
        .reduce(BigDecimal.ZERO, BigDecimal::add);
}

/**
 * 查询商家翻译成本汇总
 */
public BigDecimal getMerchantTranslationCost(Integer merId, Date startDate, Date endDate) {
    // 查询商家的所有商品ID
    List<Integer> productIds = storeProductMapper.selectList(
        new LambdaQueryWrapper<StoreProduct>()
            .eq(StoreProduct::getMerId, merId)
            .select(StoreProduct::getId)
    ).stream()
        .map(StoreProduct::getId)
        .collect(Collectors.toList());
    
    if (productIds.isEmpty()) {
        return BigDecimal.ZERO;
    }
    
    // 查询这些商品的翻译成本
    List<Translation> records = translationMapper.selectList(
        new LambdaQueryWrapper<Translation>()
            .eq(Translation::getRelatedType, "product")
            .in(Translation::getRelatedId, productIds)
            .between(Translation::getCreateTime, startDate, endDate)
            .eq(Translation::getStatus, 1)
    );
    
    return records.stream()
        .map(Translation::getCost)
        .reduce(BigDecimal.ZERO, BigDecimal::add);
}
```

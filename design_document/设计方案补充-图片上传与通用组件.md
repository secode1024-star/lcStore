# 设计方案补充 - 图片上传与通用组件

## 1. 图片上传接口设计

### 1.1 图片上传接口

**接口路径**: `POST /api/common/upload/image`

**接口说明**: 上传图片到阿里云OSS

**请求方式**: `multipart/form-data`

**请求参数**:
| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|
| file | File | 是 | 图片文件 |
| type | String | 否 | 图片类型（main-主图，slider-轮播图，detail-详情图，avatar-头像，banner-横幅） |

**文件限制**:
- 支持格式：jpg、jpeg、png、gif
- 单个文件最大：5MB
- 图片尺寸建议：
  - 主图：800x800px
  - 轮播图：1200x600px
  - 详情图：800px宽度
  - 头像：200x200px
  - Banner：1920x600px

**响应示例**:
```json
{
  "code": 200,
  "message": "上传成功",
  "data": {
    "url": "https://your-oss-bucket.oss-cn-beijing.aliyuncs.com/product/2026/05/18/abc123.jpg",
    "name": "abc123.jpg",
    "size": 102400,
    "type": "image/jpeg"
  }
}
```

**错误响应**:
```json
{
  "code": 400,
  "message": "文件格式不支持，仅支持jpg、jpeg、png、gif格式"
}
```

## 3. 前端图片上传组件

### 3.1 单图上传组件（ImageUpload.vue）

```vue
<template>
  <div class="image-upload">
    <el-upload
      :action="uploadUrl"
      :headers="headers"
      :show-file-list="false"
      :before-upload="beforeUpload"
      :on-success="handleSuccess"
      :on-error="handleError"
      accept="image/jpeg,image/jpg,image/png,image/gif"
    >
      <div v-if="imageUrl" class="image-preview">
        <img :src="imageUrl" alt="preview" />
        <div class="image-actions">
          <i class="el-icon-zoom-in" @click.stop="handlePreview"></i>
          <i class="el-icon-delete" @click.stop="handleRemove"></i>
        </div>
      </div>
      <div v-else class="upload-placeholder">
        <i class="el-icon-plus"></i>
        <div class="upload-text">{{ $t('common.uploadImage') }}</div>
      </div>
    </el-upload>

    <!-- 图片预览对话框 -->
    <el-dialog :visible.sync="previewVisible" append-to-body>
      <img :src="imageUrl" style="width: 100%;" />
    </el-dialog>
  </div>
</template>

<script>
import { getToken } from '@/utils/auth'

export default {
  name: 'ImageUpload',
  props: {
    value: {
      type: String,
      default: ''
    },
    type: {
      type: String,
      default: 'main'
    },
    maxSize: {
      type: Number,
      default: 5 // MB
    }
  },
  data() {
    return {
      imageUrl: this.value,
      previewVisible: false,
      uploadUrl: process.env.VUE_APP_BASE_API + '/api/common/upload/image',
      headers: {
        Authorization: 'Bearer ' + getToken()
      }
    }
  },
  watch: {
    value(val) {
      this.imageUrl = val
    }
  },
  methods: {
    beforeUpload(file) {
      // 验证文件类型
      const isImage = /^image\/(jpeg|jpg|png|gif)$/.test(file.type)
      if (!isImage) {
        this.$message.error(this.$t('validation.imageFormat'))
        return false
      }

      // 验证文件大小
      const isLtMaxSize = file.size / 1024 / 1024 < this.maxSize
      if (!isLtMaxSize) {
        this.$message.error(this.$t('validation.imageSize', { size: this.maxSize }))
        return false
      }

      return true
    },

    handleSuccess(response) {
      if (response.code === 200) {
        this.imageUrl = response.data.url
        this.$emit('input', this.imageUrl)
        this.$message.success(this.$t('message.uploadSuccess'))
      } else {
        this.$message.error(response.message)
      }
    },

    handleError(error) {
      this.$message.error(this.$t('message.uploadFailed'))
      console.error('上传失败', error)
    },

    handlePreview() {
      this.previewVisible = true
    },

    handleRemove() {
      this.$confirm(this.$t('message.confirmDelete'), this.$t('common.confirm'), {
        confirmButtonText: this.$t('common.confirm'),
        cancelButtonText: this.$t('common.cancel'),
        type: 'warning'
      }).then(() => {
        this.imageUrl = ''
        this.$emit('input', '')
        this.$message.success(this.$t('message.deleteSuccess'))
      }).catch(() => {})
    }
  }
}
</script>
```

### 3.2 多图上传组件（MultiImageUpload.vue）

```vue
<template>
  <div class="multi-image-upload">
    <el-upload
      :action="uploadUrl"
      :headers="headers"
      :file-list="fileList"
      :limit="limit"
      :before-upload="beforeUpload"
      :on-success="handleSuccess"
      :on-error="handleError"
      :on-remove="handleRemove"
      :on-exceed="handleExceed"
      list-type="picture-card"
      accept="image/jpeg,image/jpg,image/png,image/gif"
    >
      <i class="el-icon-plus"></i>
    </el-upload>

    <div v-if="showTip" class="upload-tip">
      {{ $t('validation.imageUploadTip', { limit, size: maxSize }) }}
    </div>
  </div>
</template>

<script>
import { getToken } from '@/utils/auth'

export default {
  name: 'MultiImageUpload',
  props: {
    value: {
      type: [String, Array],
      default: ''
    },
    type: {
      type: String,
      default: 'slider'
    },
    limit: {
      type: Number,
      default: 5
    },
    maxSize: {
      type: Number,
      default: 5
    },
    showTip: {
      type: Boolean,
      default: true
    }
  },
  data() {
    return {
      fileList: [],
      uploadUrl: process.env.VUE_APP_BASE_API + '/api/common/upload/image',
      headers: {
        Authorization: 'Bearer ' + getToken()
      }
    }
  },
  watch: {
    value: {
      handler(val) {
        if (val) {
          const urls = typeof val === 'string' ? val.split(',') : val
          this.fileList = urls.map((url, index) => ({
            uid: index,
            url: url
          }))
        } else {
          this.fileList = []
        }
      },
      immediate: true
    }
  },
  methods: {
    beforeUpload(file) {
      const isImage = /^image\/(jpeg|jpg|png|gif)$/.test(file.type)
      if (!isImage) {
        this.$message.error(this.$t('validation.imageFormat'))
        return false
      }

      const isLtMaxSize = file.size / 1024 / 1024 < this.maxSize
      if (!isLtMaxSize) {
        this.$message.error(this.$t('validation.imageSize', { size: this.maxSize }))
        return false
      }

      return true
    },

    handleSuccess(response, file, fileList) {
      if (response.code === 200) {
        file.url = response.data.url
        this.updateValue(fileList)
        this.$message.success(this.$t('message.uploadSuccess'))
      } else {
        this.$message.error(response.message)
      }
    },

    handleError(error) {
      this.$message.error(this.$t('message.uploadFailed'))
      console.error('上传失败', error)
    },

    handleRemove(file, fileList) {
      this.updateValue(fileList)
    },

    handleExceed() {
      this.$message.warning(this.$t('validation.imageLimit', { limit: this.limit }))
    },

    updateValue(fileList) {
      const urls = fileList.map(file => file.url).filter(url => url)
      const value = urls.join(',')
      this.$emit('input', value)
    }
  }
}
</script>

<style scoped>
.upload-tip {
  margin-top: 10px;
  font-size: 12px;
  color: #909399;
}
</style>
```

```vue
<script>
export default {
  name: 'Footer',
  data() {
    return {
      contactInfo: {
        email: '',
        phone: ''
      },
      contactAddressZh: '',
      contactAddressEn: '',
      contactAddressRu: '',
      contactAddressAr: '',
      aboutUsZh: '',
      aboutUsEn: '',
      aboutUsRu: '',
      aboutUsAr: '',
      copyrightZh: '',
      copyrightEn: '',
      copyrightRu: '',
      copyrightAr: ''
    }
  },
  computed: {
    currentLanguage() {
      return this.$i18n.locale
    },
    contactAddress() {
      const langMap = {
        zh: this.contactAddressZh,
        en: this.contactAddressEn,
        ru: this.contactAddressRu,
        ar: this.contactAddressAr
      }
      return langMap[this.currentLanguage] || this.contactAddressZh
    },
    aboutUsText() {
      const langMap = {
        zh: this.aboutUsZh,
        en: this.aboutUsEn,
        ru: this.aboutUsRu,
        ar: this.aboutUsAr
      }
      return langMap[this.currentLanguage] || this.aboutUsZh
    },
    copyrightText() {
      const langMap = {
        zh: this.copyrightZh,
        en: this.copyrightEn,
        ru: this.copyrightRu,
        ar: this.copyrightAr
      }
      return langMap[this.currentLanguage] || this.copyrightZh
    }
  },
  mounted() {
    this.fetchFooterInfo()
  },
  methods: {
    async fetchFooterInfo() {
      try {
        // 从系统配置获取页脚信息
        const res = await this.$api.getSystemConfig()
        const config = res.data
        
        this.contactInfo.email = config.contactEmail
        this.contactInfo.phone = config.contactPhone
        this.contactAddressZh = config.contactAddressZh
        this.contactAddressEn = config.contactAddressEn
        this.contactAddressRu = config.contactAddressRu
        this.contactAddressAr = config.contactAddressAr
        this.aboutUsZh = config.aboutUsZh
        this.aboutUsEn = config.aboutUsEn
        this.aboutUsRu = config.aboutUsRu
        this.aboutUsAr = config.aboutUsAr
        this.copyrightZh = config.copyrightZh
        this.copyrightEn = config.copyrightEn
        this.copyrightRu = config.copyrightRu
        this.copyrightAr = config.copyrightAr
      } catch (error) {
        console.error('获取页脚信息失败', error)
      }
    }
  }
}
</script>
```

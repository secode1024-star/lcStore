### 6.1 翻译状态标签组件（TranslationStatusTag.vue）

**用途**: 显示翻译状态的标签

**Props**:
```javascript
{
  status: {
    type: Number,
    required: true
    // 0-未翻译，1-翻译中，2-翻译完成，3-翻译失败
  }
}
```

**实现**:
```vue
<template>
  <el-tag :type="tagType" size="small">
    <i :class="iconClass"></i>
    {{ statusText }}
  </el-tag>
</template>

<script>
export default {
  name: 'TranslationStatusTag',
  props: {
    status: {
      type: Number,
      required: true
    }
  },
  computed: {
    statusText() {
      const statusMap = {
        0: '未翻译',
        1: '翻译中',
        2: '翻译完成',
        3: '翻译失败'
      }
      return statusMap[this.status] || '未知'
    },
    tagType() {
      const typeMap = {
        0: 'info',
        1: 'warning',
        2: 'success',
        3: 'danger'
      }
      return typeMap[this.status] || 'info'
    },
    iconClass() {
      const iconMap = {
        0: 'el-icon-minus',
        1: 'el-icon-loading',
        2: 'el-icon-check',
        3: 'el-icon-close'
      }
      return iconMap[this.status] || ''
    }
  }
}
</script>
```

### 6.4 未读消息提醒组件（UnreadMessageBadge.vue）

**用途**: 顶部导航栏显示未读消息数量

**实现**:
```vue
<template>
  <div class="unread-message-badge" @click="handleClick">
    <el-badge :value="unreadCount" :max="99" :hidden="unreadCount === 0">
      <i class="el-icon-message" style="font-size: 20px;"></i>
    </el-badge>
    
    <!-- 未读消息弹出框 -->
    <el-popover
      v-model="popoverVisible"
      placement="bottom"
      width="300"
      trigger="click"
    >
      <div v-loading="loading">
        <div v-if="messages.length > 0">
          <div
            v-for="msg in messages"
            :key="msg.id"
            class="message-item"
            @click="goToDetail(msg.id)"
          >
            <message-type-tag :type="msg.type" />
            <p class="message-title">{{ msg.title }}</p>
            <p class="message-time">{{ msg.createTime }}</p>
          </div>
          <div class="view-all" @click="goToList">
            查看全部消息
          </div>
        </div>
        <div v-else class="no-message">
          暂无未读消息
        </div>
      </div>
      
      <el-badge
        slot="reference"
        :value="unreadCount"
        :max="99"
        :hidden="unreadCount === 0"
      >
        <i class="el-icon-message" style="font-size: 20px; cursor: pointer;"></i>
      </el-badge>
    </el-popover>
  </div>
</template>

<script>
import { getUnreadCount, getMessageList } from '@/api/message'
import MessageTypeTag from './MessageTypeTag'

export default {
  name: 'UnreadMessageBadge',
  components: { MessageTypeTag },
  data() {
    return {
      unreadCount: 0,
      messages: [],
      popoverVisible: false,
      loading: false,
      timer: null
    }
  },
  mounted() {
    this.fetchUnreadCount()
    // 每30秒刷新一次未读数量
    this.timer = setInterval(() => {
      this.fetchUnreadCount()
    }, 30000)
  },
  beforeDestroy() {
    if (this.timer) {
      clearInterval(this.timer)
    }
  },
  methods: {
    async fetchUnreadCount() {
      try {
        const res = await getUnreadCount()
        this.unreadCount = res.data
      } catch (error) {
        console.error('获取未读消息数量失败', error)
      }
    },
    async fetchMessages() {
      this.loading = true
      try {
        const res = await getMessageList({
          isRead: 0,
          pageNum: 1,
          pageSize: 5
        })
        this.messages = res.data.list
      } catch (error) {
        this.$message.error('获取消息列表失败')
      } finally {
        this.loading = false
      }
    },
    handleClick() {
      if (this.popoverVisible) {
        this.fetchMessages()
      }
    },
    goToDetail(id) {
      this.popoverVisible = false
      this.$router.push(`/message/detail/${id}`)
    },
    goToList() {
      this.popoverVisible = false
      this.$router.push('/message/list')
    }
  }
}
</script>

<style scoped>
```

### 8.3 API请求封装

```javascript
// utils/request.js

import axios from 'axios'
import { Message } from 'element-ui'
import store from '@/store'
import { getToken } from '@/utils/auth'

// 创建axios实例
const service = axios.create({
  baseURL: process.env.VUE_APP_BASE_API,
  timeout: 15000
})

// 请求拦截器
service.interceptors.request.use(
  config => {
    // 添加token
    if (store.getters.token) {
      config.headers['Authorization'] = 'Bearer ' + getToken()
    }
    return config
  },
  error => {
    console.error(error)
    return Promise.reject(error)
  }
)

// 响应拦截器
service.interceptors.response.use(
  response => {
    const res = response.data
    
    // 根据后端返回的code判断
    if (res.code !== 200) {
      Message({
        message: res.message || 'Error',
        type: 'error',
        duration: 5 * 1000
      })
      
      // 401: 未登录
      if (res.code === 401) {
        store.dispatch('user/logout').then(() => {
          location.reload()
        })
      }
      
      return Promise.reject(new Error(res.message || 'Error'))
    } else {
      return res
    }
  },
  error => {
    console.error('err' + error)
    Message({
      message: error.message,
      type: 'error',
      duration: 5 * 1000
    })
    return Promise.reject(error)
  }
)

export default service
```

### 8.4 权限控制

```javascript
// 路由守卫
router.beforeEach(async(to, from, next) => {
  // 获取token
  const hasToken = getToken()

  if (hasToken) {
    if (to.path === '/login') {
      // 已登录，跳转到首页
      next({ path: '/' })
    } else {
      // 检查是否有用户信息
      const hasUserInfo = store.getters.name
      if (hasUserInfo) {
        next()
      } else {
        try {
          // 获取用户信息
          await store.dispatch('user/getInfo')
          next()
        } catch (error) {
          // 获取用户信息失败，清除token并跳转到登录页
          await store.dispatch('user/resetToken')
          Message.error(error || 'Has Error')
          next(`/login?redirect=${to.path}`)
        }
      }
    }
  } else {
    // 未登录
    if (whiteList.indexOf(to.path) !== -1) {
      // 在白名单中，直接进入
      next()
    } else {
      // 不在白名单中，跳转到登录页
      next(`/login?redirect=${to.path}`)
    }
  }
})
```

# 设计方案补充 - 用户认证（登录注册）

## 1. 业务逻辑

### 1.1 用户模式

**可选登录模式**:
- **游客模式**: 可以浏览商品、提交询价（需填写联系信息）
- **登录模式**: 可以保存个人信息、查看询价历史、使用意向清单

### 1.2 注册方式

支持两种注册方式：
1. **邮箱注册**: 邮箱 + 密码
2. **手机号注册**: 手机号 + 密码

### 1.3 登录方式

支持两种登录方式：
1. **邮箱登录**: 邮箱 + 密码
2. **手机号登录**: 手机号 + 密码

## 3. API接口设计

### 3.1 用户注册

**接口路径**: `POST /api/user/register`

**接口说明**: 用户注册

**请求参数**:
```json
{
  "registerType": "email",  // 注册类型：email-邮箱，phone-手机号
  "account": "user@example.com",  // 账号（邮箱或手机号）
  "password": "123456",  // 密码
  "confirmPassword": "123456",  // 确认密码
  "nickname": "用户昵称",  // 昵称（可选）
  "language": "zh"  // 首选语言
}
```

**参数验证**:
- account: 必填，邮箱格式或手机号格式
- password: 必填，6-20位，包含字母和数字
- confirmPassword: 必填，与password一致
- nickname: 可选，2-20个字符

**响应示例**:
```json
{
  "code": 200,
  "message": "注册成功",
  "data": {
    "token": "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9...",
    "userInfo": {
      "uid": 1,
      "account": "user@example.com",
      "nickname": "用户昵称",
      "avatar": "",
      "preferredLanguage": "zh"
    }
  }
}
```

| 参数名 | 类型 | 必填 | 说明 |
|--------|------|------|------|

```vue
      </div>

        ref="loginForm"
        :model="form"
        :rules="rules"
        class="login-form"

        </el-form-item>
        
        <el-form-item prop="password">
          <el-input
            v-model="form.password"
            type="password"

          />

            {{ $t('user.registerNow') }}

      </el-form>
    </div>
  </div>

<script>
import { login } from '@/api/user'

export default {
  name: 'Login',
  data() {
    return {
      form: {
        account: '',
        password: ''
      },
      loading: false,
      rules: {
        account: [
          { required: true, message: this.$t('validation.required'), trigger: 'blur' }
        ],
        password: [
          { required: true, message: this.$t('validation.required'), trigger: 'blur' },
          { min: 6, max: 20, message: this.$t('validation.passwordLength'), trigger: 'blur' }
        ]
      }
    }
  },
  methods: {
    handleLogin() {
      this.$refs.loginForm.validate(async (valid) => {
        if (!valid) return
        
        this.loading = true
        try {
          const res = await login(this.form)
          
          // 保存token和用户信息
          this.$store.dispatch('user/login', res.data)
          
          this.$message.success(this.$t('user.loginSuccess'))
          
          // 跳转到来源页面或首页
          const redirect = this.$route.query.redirect || '/'
          this.$router.push(redirect)
        } catch (error) {
          this.$message.error(error.message || this.$t('user.loginFailed'))
        } finally {
          this.loading = false
        }
      })
    }
  }
}
</script>
```

**实现代码**:
```vue
<template>
  <div class="register-page">
    <div class="register-container">
      <div class="logo">
        <img src="@/assets/logo.png" alt="Logo" />
      </div>
      
      <h2>{{ $t('common.register') }}</h2>
      
      <el-form
        ref="registerForm"
        :model="form"
        :rules="rules"
        class="register-form"
      >
        <el-form-item :label="$t('user.registerType')">
          <el-radio-group v-model="form.registerType">
            <el-radio label="email">{{ $t('user.emailRegister') }}</el-radio>
            <el-radio label="phone">{{ $t('user.phoneRegister') }}</el-radio>
          </el-radio-group>
        </el-form-item>
        
        <el-form-item prop="account">
          <el-input
            v-model="form.account"
            :placeholder="accountPlaceholder"
            prefix-icon="el-icon-user"
            @blur="checkAccount"
          />
        </el-form-item>
        
        <el-form-item prop="password">
          <el-input
            v-model="form.password"
            type="password"
            :placeholder="$t('user.passwordPlaceholder')"
            prefix-icon="el-icon-lock"
            show-password
          />
        </el-form-item>
        
        <el-form-item prop="confirmPassword">
          <el-input
            v-model="form.confirmPassword"
            type="password"
            :placeholder="$t('user.confirmPasswordPlaceholder')"
            prefix-icon="el-icon-lock"
            show-password
          />
        </el-form-item>
        
        <el-form-item prop="nickname">
          <el-input
            v-model="form.nickname"
            :placeholder="$t('user.nicknamePlaceholder')"
            prefix-icon="el-icon-user"
          />
        </el-form-item>
        
        <el-form-item>
          <el-button
            type="primary"
            :loading="loading"
            class="register-button"
            @click="handleRegister"
          >
            {{ $t('common.register') }}
          </el-button>
        </el-form-item>
        
        <div class="register-footer">
          <span>{{ $t('user.hasAccount') }}</span>
          <router-link to="/login">
            {{ $t('user.loginNow') }}
          </router-link>
        </div>
      </el-form>
    </div>
  </div>
</template>

<script>
import { register, checkAccountExists } from '@/api/user'

export default {
  name: 'Register',
  data() {
    // 验证确认密码
    const validateConfirmPassword = (rule, value, callback) => {
      if (value !== this.form.password) {
        callback(new Error(this.$t('validation.passwordNotMatch')))
      } else {
        callback()
      }
    }
    
    // 验证账号格式
    const validateAccount = (rule, value, callback) => {
      if (this.form.registerType === 'email') {
        const emailReg = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
        if (!emailReg.test(value)) {
          callback(new Error(this.$t('validation.emailFormat')))
        } else {
          callback()
        }
      } else {
        const phoneReg = /^1[3-9]\d{9}$/
        if (!phoneReg.test(value)) {
          callback(new Error(this.$t('validation.phoneFormat')))
        } else {
          callback()
        }
      }
    }
    
    return {
      form: {
        registerType: 'email',
        account: '',
        password: '',
        confirmPassword: '',
        nickname: '',
        language: this.$i18n.locale
      },
      loading: false,
      rules: {
        account: [
          { required: true, message: this.$t('validation.required'), trigger: 'blur' },
          { validator: validateAccount, trigger: 'blur' }
        ],
        password: [
          { required: true, message: this.$t('validation.required'), trigger: 'blur' },
          { min: 6, max: 20, message: this.$t('validation.passwordLength'), trigger: 'blur' }
        ],
        confirmPassword: [
          { required: true, message: this.$t('validation.required'), trigger: 'blur' },
          { validator: validateConfirmPassword, trigger: 'blur' }
        ],
        nickname: [
          { min: 2, max: 20, message: this.$t('validation.nicknameLength'), trigger: 'blur' }
        ]
      }
    }
  },
  computed: {
    accountPlaceholder() {
      return this.form.registerType === 'email'
        ? this.$t('user.emailPlaceholder')
        : this.$t('user.phonePlaceholder')
    }
  },
  methods: {
    async checkAccount() {
      if (!this.form.account) return
      
      try {
        const res = await checkAccountExists({ account: this.form.account })
        if (res.data.exists) {
          this.$message.warning(this.$t('user.accountExists'))
        }
      } catch (error) {
        console.error('检查账号失败', error)
      }
    },
    
    handleRegister() {
      this.$refs.registerForm.validate(async (valid) => {
        if (!valid) return
        
        this.loading = true
        try {
          const res = await register(this.form)
          
          // 保存token和用户信息
          this.$store.dispatch('user/login', res.data)
          
          this.$message.success(this.$t('user.registerSuccess'))
          
          // 跳转到首页
          this.$router.push('/')
        } catch (error) {
          this.$message.error(error.message || this.$t('user.registerFailed'))
        } finally {
          this.loading = false
        }
      })
    }
  }
}
</script>
```

## 6. Vuex Store

```javascript
// store/modules/user.js
import { login, logout, getUserInfo } from '@/api/user'
import { getToken, setToken, removeToken } from '@/utils/auth'

const state = {
  token: getToken(),
  userInfo: null
}

const mutations = {
  SET_TOKEN(state, token) {
    state.token = token
  },
  SET_USER_INFO(state, userInfo) {
    state.userInfo = userInfo
  }
}

const actions = {
  // 登录
  login({ commit }, data) {
    commit('SET_TOKEN', data.token)
    commit('SET_USER_INFO', data.userInfo)
    setToken(data.token)
  },
  
  // 获取用户信息
  async getInfo({ commit }) {
    try {
      const res = await getUserInfo()
      commit('SET_USER_INFO', res.data)
      return res.data
    } catch (error) {
      throw error
    }
  },
  
  // 退出登录
  async logout({ commit }) {
    try {
      await logout()
      commit('SET_TOKEN', '')
      commit('SET_USER_INFO', null)
      removeToken()
    } catch (error) {
      throw error
    }
  }
}

export default {
  namespaced: true,
  state,
  mutations,
  actions
}
```

## 8. 注意事项

1. **密码加密**: 前端传输密码时建议使用MD5或SHA256加密，后端再次加密存储
2. **Token管理**: 使用JWT Token，设置合理的过期时间（如7天）
3. **账号唯一性**: 邮箱和手机号都需要保证唯一性
4. **验证码**: 生产环境建议添加图形验证码或短信验证码
5. **第三方登录**: 后续可扩展支持微信、Google等第三方登录
6. **找回密码**: 后续可添加找回密码功能
